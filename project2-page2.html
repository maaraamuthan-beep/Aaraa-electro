Obstacle Avoiding Robot with Servo Motor - Arduino Code & Build Video

Arduino Code (Obstacle Avoiding with Servo)

Copy this code into the Arduino IDE (Requires Servo.h library). 
// Obstacle Avoiding Robot with Servo Motor - Aaraa Electro
// Ultrasonic Sensor (HC-SR04), Servo Motor, and L298N Motor Driver

#include <Servo.h>

const int trigPin = A1;
const int echoPin = A2;
const int servoPin = 3;

const int ENA = 5;
const int IN1 = 6;
const int IN2 = 7;
const int IN3 = 8;
const int IN4 = 9;
const int ENB = 10;

const int speedValue = 160;
const int safeDistance = 25; // cm

Servo scanServo;

long readDistanceCM() {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  long duration = pulseIn(echoPin, HIGH, 30000);
  if (duration == 0) return 999;
  return (duration * 0.034) / 2;
}

void setup() {
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);

  pinMode(ENA, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT);
  pinMode(IN4, OUTPUT);
  pinMode(ENB, OUTPUT);

  scanServo.attach(servoPin);
  scanServo.write(90); // Look straight ahead
  delay(500);
}

void loop() {
  long frontDistance = readDistanceCM();

  if (frontDistance > safeDistance) {
    moveForward();
  } else {
    stopRobot();
    delay(200);
    moveBackward();
    delay(300);
    stopRobot();
    delay(150);

    // Scan Right
    scanServo.write(30);
    delay(400);
    long rightDistance = readDistanceCM();

    // Scan Left
    scanServo.write(150);
    delay(500);
    long leftDistance = readDistanceCM();

    // Return Servo to Center
    scanServo.write(90);
    delay(300);

    if (rightDistance > leftDistance) {
      turnRight();
      delay(450);
    } else {
      turnLeft();
      delay(450);
    }
    stopRobot();
  }
}

void moveForward() {
  analogWrite(ENA, speedValue);
  analogWrite(ENB, speedValue);
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void moveBackward() {
  analogWrite(ENA, speedValue);
  analogWrite(ENB, speedValue);
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}

void turnLeft() {
  analogWrite(ENA, speedValue);
  analogWrite(ENB, speedValue);
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}

void turnRight() {
  analogWrite(ENA, speedValue);
  analogWrite(ENB, speedValue);
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}

void stopRobot() {
  analogWrite(ENA, 0);
  analogWrite(ENB, 0);
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}
